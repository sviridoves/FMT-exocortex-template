# Проверки Day Open по умолчанию (шаблон)

> Исполняются `scripts/day-open-checks-runner.sh` на шаге 5 конвейера открытия дня,
> ДО коммита DayPlan. Каждый ```bash```-блок = одна проверка; ненулевой выход
> блокирует коммит. Переменная `$FILE` — путь к сегодняшнему DayPlan.
>
> Это ДЕФОЛТНЫЙ набор шаблона (WP-529 Ф7: без него шаг Checks у свежей установки
> честно говорил «проверять нечего» и деградировал). Он проверяет только
> универсальные инварианты, верные для любой установки. Свои проверки добавляйте
> отдельными файлами `day-open.checks.<имя>.md` рядом — раннер подхватит все.
>
> Файл-проверка, адресованный АГЕНТУ (проза, без bash-блоков), объявляет это
> строкой `<!-- executor: agent -->` в первых 15 строках — раннер такой файл
> пропускает как «неприменимо», а не валит как ошибку разметки (issue #546).
> Файл без декларации и без bash-блоков остаётся ошибкой: это может быть и
> опечатка в ```bash-заборе, и повреждённый файл.

### 🔴 ПРОВЕРКА: DayPlan существует и похож на план

> Файл должен быть непустым markdown-документом с заголовком первого уровня.
> Пустой или обрубленный файл — признак упавшей генерации; коммитить нечего.

```bash
[ -s "$FILE" ] || { echo "  ❌ DayPlan пуст или отсутствует: $FILE"; exit 1; }
FIRST_H1=$(grep -m1 '^# ' "$FILE" || true)
[ -n "$FIRST_H1" ] || { echo "  ❌ В DayPlan нет ни одного '# '-заголовка первого уровня"; exit 1; }
LINES=$(wc -l < "$FILE" | tr -d ' ')
[ "$LINES" -ge 5 ] || { echo "  ❌ DayPlan подозрительно короткий ($LINES строк) — похоже на оборванную генерацию"; exit 1; }
echo "  ✅ DayPlan существует, заголовок на месте, $LINES строк"
```

### 🔴 ПРОВЕРКА: дата в имени файла согласована с заголовком

> Имя файла (`DayPlan YYYY-MM-DD.md`) и дата в заголовке H1 (если она там есть)
> обязаны совпадать — расхождение означает, что проверяется не тот план
> (класс бага head-1/tail-1 из авторского checks-файла, 2026-07-11).

```bash
FNAME_DATE=$(basename "$FILE" | grep -oE '[0-9]{4}-[0-9]{2}-[0-9]{2}' | head -1)
[ -n "$FNAME_DATE" ] || { echo "  ✅ Имя файла без даты — проверка неприменима"; exit 0; }
FIRST_H1=$(grep -m1 '^# ' "$FILE" || true)
H1_DATE=$(printf '%s\n' "$FIRST_H1" | grep -oE '[0-9]{4}-[0-9]{2}-[0-9]{2}' | head -1 || true)
if [ -n "$H1_DATE" ] && [ "$H1_DATE" != "$FNAME_DATE" ]; then
  echo "  ❌ Дата в заголовке ($H1_DATE) не совпадает с именем файла ($FNAME_DATE) — проверяется не тот план?"
  exit 1
fi
echo "  ✅ Дата согласована: $FNAME_DATE"
```

### 🟡 ПРОВЕРКА: секция «Требует внимания» присутствует

> Предупреждение, не блок: детерминированные шаги конвейера (4.55-4.57)
> дописывают находки именно в эту секцию — без неё находки молча теряются.

```bash
if grep -q "Требует внимания" "$FILE"; then
  echo "  ✅ Секция «Требует внимания» на месте"
else
  echo "  🟡 В DayPlan нет секции «Требует внимания» — находки детерминированных шагов (приоритеты, ошибки закрытия, версия) не смогут записаться"
fi
```

### 🔴 ПРОВЕРКА: номера WP в плане дня существуют и открыты

> LLM-fill склонен придумывать номера РП (галлюцинация колонок '#'): ставить
> закрытому или чужому проекту номер из REGISTRY по соседству. Каждый WP-N из
> таблицы «План на сегодня» обязан иметь открытый контекст `inbox/WP-N/`
> (status pending/in_progress/blocked). Номера без контекста блокируют коммит;
> ad-hoc задачи должны идти с '—', а не с выдуманным номером.
> Источник дефекта: Day Open 2026-10-20 (КИСУ получил WP-8 = «Алроса_СУД»).

```bash
# Сканируем ТОЛЬКО колонку '#' таблицы плана (строки '| ... |'), не прозу.
PLAN_TABLE=$(awk '/План на сегодня/{f=1} f&&/<\/details>/{exit} f
  /^##[^#]*Carry-over/{f=0}' "$FILE" | grep '^|')
BAD=""
REPO_ROOT="$(cd "$(dirname "$FILE")/.." && pwd)"
for n in $(printf '%s\n' "$PLAN_TABLE" | awk -F'|' '{print $4}' | grep -oE 'WP-[0-9]+' | sort -u); do
  id=${n#WP-}
  ctx_file="$REPO_ROOT/inbox/$n/$n.md"
  if [ ! -f "$ctx_file" ]; then
    # legacy-раскладка inbox/WP-N.md тоже считается контекстом
    [ -f "$REPO_ROOT/inbox/$n.md" ] || { BAD="$BAD $n(нет контекста)"; continue; }
    ctx_file="$REPO_ROOT/inbox/$n.md"
  fi
  st=$(grep -m1 '^status:' "$ctx_file" | sed 's/status:[[:space:]]*//' | tr -d "\"'" | head -1)
  case "$st" in
    done|closed|archived) BAD="$BAD $n(status=$st)" ;;
  esac
done
if [ -n "$BAD" ]; then
  echo "  ❌ Колонка '#' плана ссылается на закрытые/несуществующие РП:$BAD — допустимы только открытые WP-ID или '—'"
  exit 1
fi
echo "  ✅ Номера WP в колонке '#' проверены (все открыты либо стоит '—')"
```
